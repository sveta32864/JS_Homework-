14. !!'false' == !!'true'
    console.log(!!'false' == !!'true');
    Виведе в консоль: true

/* Оператор !! (подвійне заперечення): це два оператори ! (заперечення), які використовуються послідовно,
 мета — перетворити будь-яке значення на булевий еквівалент.
  !!'false' перший оператор ! перетворює рядок 'false' на false. Другий оператор ! перетворює false на true.
  !!'true' перший оператор ! перетворює рядок 'true' на false.Другий оператор ! перетворює false на true.
   Оператор == порівнює два булеві значення true і true, результат true
*/

15. 0 || '0' && 1
    var result = 0 || '0' && 1;
   console.log(result); 
   Виведе в консоль: 1

/*   '0' && 1: логічний оператор && повертає перше хибне значення (falsy) або останнє істинне значення (truthy).
      Число 0 є хибним значенням (falsy),число 1 є істинним значенням (truthy),
      результатом буде перше істинне значення 1
      0 || 1: логічний оператор || повертає перше істинне значення (truthy) або останнє хибне значення (falsy).
      Число 0 є хибним значенням (falsy),число 1 є істинним значенням (truthy), результатом буде перше істинне значення 1.
*/
